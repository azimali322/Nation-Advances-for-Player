suez_canal_advance = {
	age = age_5_absolutism

	potential = {
		OR = {
			AND = {
				has_variable = hafp_enabled
				OR = {
					has_variable = hafp_all_advances_enabled
					has_variable = hafp_g_age_age_5_absolutism
					has_variable = hafp_g_cont_africa
					has_variable = hafp_g_cont_asia
					has_variable = hafp_g_cont_europe
				}
			}
			AND = {
				has_discovered = location:suez
				original_capital ?= {
				OR = {
				continent = continent:europe
				continent = continent:asia
				continent = continent:africa
				}
				}
			}
		}
	}

	allow = {
		owns_or_non_sovereign_subject_owns = location:suez
	}

	allow_children = no
}

panama_canal_advance = {
	age = age_5_absolutism

	potential = {
		OR = {
			AND = {
				has_variable = hafp_enabled
				OR = {
					has_variable = hafp_all_advances_enabled
					has_variable = hafp_g_age_age_5_absolutism
					has_variable = hafp_g_cont_africa
					has_variable = hafp_g_cont_asia
					has_variable = hafp_g_cont_europe
				}
			}
			AND = {
				has_discovered = location:panama
				original_capital ?= {
				NOT = { continent = continent:oceania }
				}
			}
		}
	}

	allow = {
		owns_or_non_sovereign_subject_owns = location:panama
	}

	allow_children = no
}
